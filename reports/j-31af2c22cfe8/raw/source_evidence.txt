   105	        tl.float32
   106	    )
   107	    xr1 = (xr1 * inv_rms * (wr1 + 1.0)).to(out_dtype).to(tl.float32)
   108	    xr2 = (xr2 * inv_rms * (wr2 + 1.0)).to(out_dtype).to(tl.float32)
   109	
   110	    if MROPE:
   111	        axis = tl.load(mrope_axis_map_ptr + rot_offs, mask=rot_mask, other=0)
   112	        pos = tl.load(
   113	            positions_ptr + axis * stride_pos_axis + token, mask=rot_mask, other=0
   114	        )
   115	    else:
   116	        pos = tl.load(positions_ptr + token)
   117	    cache_off = pos.to(tl.int64) * stride_cos_t
   118	    cos = tl.load(
   119	        cos_sin_cache_ptr + cache_off + rot_offs, mask=rot_mask, other=0.0
   120	    ).to(tl.float32)
   121	    sin = tl.load(
   122	        cos_sin_cache_ptr + cache_off + HALF_ROTARY + rot_offs, mask=rot_mask, other=0.0
   123	    ).to(tl.float32)
   124	    tl.store(out_base + rot_offs, (xr1 * cos - xr2 * sin), mask=rot_mask)
   125	    tl.store(out_base + HALF_ROTARY + rot_offs, (xr2 * cos + xr1 * sin), mask=rot_mask)
   126	
   127	    # Gate copy (Q heads only)
   128	    if HAS_GATE and not is_k:
   129	        gate_in = in_base + HEAD_DIM
   130	        gate_out = gate_out_ptr + token * stride_gate_t + local_head * HEAD_DIM
   131	        g = tl.load(gate_in + head_offs, mask=head_mask, other=0.0)
   132	        tl.store(gate_out + head_offs, g, mask=head_mask)
   133	
   134	    # PDL: signal dependent kernels (attention/allreduce) can start early.
   135	    # Only available on NVIDIA Hopper+ (sm_90+); guarded for AMD/other backends.
   136	    if ENABLE_PDL:
   137	        tl.extra.cuda.gdc_launch_dependents()

   156	
   157	    Args:
   158	        q_gate: [T, num_q_heads * (1 + has_gate) * head_dim] — interleaved Q+Gate if has_gate

   166	    assert positions.dim() in (1, 2), f"want [T] or [3, T], got {positions.shape}"

   169	    if mrope:
   170	        assert positions.shape[0] == 3 and positions.stride(1) == 1, (
   171	            f"want [3, T] contiguous over T, got {positions.shape} "
   172	            f"stride {positions.stride()}"
   173	        )
   174	        lanes = rotary_dim // 2
   175	        assert mrope_axis_map.shape == (lanes,), f"want one axis per lane ({lanes})"
   176	    T = q_gate.shape[0]
   177	    q_size = num_q_heads * head_dim
   178	    kv_size = num_kv_heads * head_dim
   179	
   180	    q_out = torch.empty(T, q_size, dtype=q_gate.dtype, device=q_gate.device)
   181	    k_out = torch.empty(T, kv_size, dtype=k.dtype, device=k.device)
   182	    gate_out = (
   183	        torch.empty(T, num_q_heads, head_dim, dtype=q_gate.dtype, device=q_gate.device)
   184	        if has_gate
   185	        else q_out
   186	    )
   187	
   188	    half_rotary = rotary_dim // 2
   189	    head_block = triton.next_power_of_2(head_dim)
   190	    rot_half_block = triton.next_power_of_2(half_rotary)
   191	
   192	    grid = (T, num_q_heads + num_kv_heads)
   193	    _fused_qk_rmsnorm_rope_gate_kernel[grid](
   194	        q_gate,
   195	        k,
   196	        q_out,
   197	        k_out,
   198	        gate_out,
   199	        q_weight,
   200	        k_weight,
   201	        cos_sin_cache,
   202	        positions,
   203	        mrope_axis_map,
   204	        q_gate.stride(0),
   205	        k.stride(0),
   206	        q_out.stride(0),
   207	        k_out.stride(0),
   208	        gate_out.stride(0),
   209	        cos_sin_cache.stride(0),
   210	        positions.stride(0),
   211	        NUM_Q_HEADS=num_q_heads,
   212	        NUM_KV_HEADS=num_kv_heads,
   213	        HEAD_DIM=head_dim,
   214	        ROTARY_DIM=rotary_dim,
   215	        HALF_ROTARY=half_rotary,
   216	        HEAD_BLOCK=head_block,
   217	        ROT_HALF_BLOCK=rot_half_block,
   218	        EPS=eps,
   219	        FP16=q_gate.dtype == torch.float16,
   220	        HAS_PASS=rotary_dim < head_dim,
