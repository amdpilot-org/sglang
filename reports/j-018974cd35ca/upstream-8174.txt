It seems the issue is b/c Llama 3.2 does not have good support for multi-image. Meta engineers also confirmed this issue in this thread: https://huggingface.co/meta-llama/Llama-3.2-11B-Vision-Instruct/discussions/43

### Code Reference ([Adapted from official wiki](https://huggingface.co/meta-llama/Llama-3.2-11B-Vision-Instruct#use-with-transformers))
```
import requests
import torch
from PIL import Image
from transformers import MllamaForConditionalGeneration, AutoProcessor

model_id = "/shared/public/elr-models/meta-llama/Llama-3.2-11B-Vision-Instruct/9eb2daaa8597bf192a8b0e73f848f3a102794df5/"

model = MllamaForConditionalGeneration.from_pretrained(
    model_id,
    torch_dtype=torch.bfloat16,
    device_map="auto"
)
processor = AutoProcessor.from_pretrained(model_id)

IMAGE_MAN_IRONING_URL = "/home/jobuser/sglang/man_ironing_on_back_of_suv.png"
IMAGE_SGL_LOGO_URL = "/home/jobuser/sglang/sgl_logo.png"
image1 = Image.open(IMAGE_MAN_IRONING_URL)
image2 = Image.open(IMAGE_SGL_LOGO_URL)

messages = [
    {"role": "user", "content": [
        {"type": "image", "modalities": "multi-images"},
        {"type": "image", "modalities": "multi-images"},
        {"type": "text", "text": "I have two very different images. They are not related at all. Please describe the first image in one sentence, and then describe the second image in another sentence."}
    ]}
]
input_text = processor.apply_chat_template(messages, add_generation_prompt=True)

inputs = processor(
    [image1, image2],
    input_text,
    add_special_tokens=False,
    return_tensors="pt"
).to(model.device)

print(inputs.input_ids)

output = model.generate(**inputs, max_new_tokens=1000)
print(processor.decode(output[0]))

```
