helper_source def can_fuse_glm5_next_qkvbfg(
    quant_config: Optional[QuantizationConfig],
    prefix: str,
    head_shard_size: int,
    tp_size: int,
) -> bool:
    if head_shard_size != tp_size:
        return False

    projection_names = (
        "qkv_proj",
        "f_a_proj",
        "f_b_proj",
        "b_proj",
        "g_a_proj",
        "g_b_proj",
    )
    return are_linear_prefixes_unquantized(
        quant_config, (f"{prefix}.{name}" for name in projection_names)
    )
