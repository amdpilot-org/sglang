    75	        select_index = self.free_slots[:need_size]
    76	        self.free_slots = self.free_slots[need_size:]
    77	        return select_index
    78	
    79	    def free(self, free_index: torch.Tensor):
    80	        if free_index.numel() == 0:
    81	            return
    82	        self.free_slots = torch.cat((self.free_slots, free_index))
    83	
    84	    def clear(self):
    85	        # Slot 0 is reserved as a dummy write target for padded tokens.
    86	        self.free_slots = torch.arange(
    87	            1, self.size + 1, dtype=torch.int64, device=self.device
    88	        )
