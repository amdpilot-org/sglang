   743	def handle_multimodal_feature_transport(server_args: Any):
   744	    """Resolve multimodal feature transport before tokenizer workers start.
   745
   746	    CUDA IPC is opt-in because its fixed pool on ``base_gpu_id`` reduces the
   747	    memory left for model/KV-cache allocations. Multi-node MNNVL deployments
   748	    may still auto-select CUDA VMM. The legacy CUDA IPC flag and environment
   749	    variable remain supported so existing deployments map to this policy.
   750	    """
   751
   752	    cfg = resolving_view(server_args)
   753	    requested_transport = cfg.mm_feature_transport
   754	    legacy_ipc_is_set = envs.SGLANG_USE_CUDA_IPC_TRANSPORT.is_set()
   755	    legacy_ipc_enabled = envs.SGLANG_USE_CUDA_IPC_TRANSPORT.get()
   756
   757	    if cfg.keep_mm_feature_on_device:
   758	        if requested_transport not in (None, "cuda_ipc"):
   759	            raise ValueError(
   760	                "--keep-mm-feature-on-device conflicts with "
   761	                f"--mm-feature-transport={requested_transport}. Use only "
   762	                "--mm-feature-transport=cuda_ipc."
   763	            )
   764	        requested_transport = "cuda_ipc"
   765	        logger.warning(
   766	            "--keep-mm-feature-on-device is deprecated; using "
   767	            "--mm-feature-transport=cuda_ipc instead."
   768	        )
   769
   770	    if requested_transport is None:
   771	        if legacy_ipc_is_set:
   772	            requested_transport = "cuda_ipc" if legacy_ipc_enabled else "cpu"
   773	            logger.warning(
   774	                "SGLANG_USE_CUDA_IPC_TRANSPORT is deprecated; use "
   775	                "--mm-feature-transport=%s instead.",
   776	                requested_transport,
   777	            )
   778	        elif cfg.encoder_only:
   779	            requested_transport = "cpu"
   780	            logger.info(
   781	                "Multimodal feature transport auto-resolved to cpu for "
   782	                "encoder-only serving; encoder outputs use "
   783	                "--encoder-transfer-backend instead."
   784	            )
   785	        elif (
   786	            model_config_of(server_args).is_multimodal
   787	            and get_platform().is_cuda
   788	            and cfg.disaggregation_mode == "null"
   789	        ):
   790	            # A full GPU pool always degrades to CPU transport per tensor.
   791	            # Keep CUDA IPC opt-in because even an idle pool consumes HBM
   792	            # that would otherwise back the KV cache. Multi-node
   793	            # auto-selection is limited to GB200/GB300 systems where the
   794	            # runtime already enables the MNNVL/IMEX communication stack.
   795	            if cfg.nnodes == 1:
   796	                requested_transport = "cpu"
   797	            elif is_mnnvl_fabric_device() and os.path.exists(
   798	                "/dev/nvidia-caps-imex-channels/channel0"
   799	            ):
   800	                from sglang.srt.model_loader.utils import (
   801	                    supports_cuda_vmm_feature_transport,
   802	                )
   803
   804	                if supports_cuda_vmm_feature_transport(model_config_of(server_args)):
   805	                    requested_transport = "cuda_vmm"
   806	                    logger.info(
   807	                        "Multimodal feature transport auto-resolved to "
   808	                        "cuda_vmm (multi-node GB200/GB300 MNNVL). Pass "
   809	                        "--mm-feature-transport=cpu to opt out."
   810	                    )
   811	                else:
   812	                    requested_transport = "cpu"
   813	                    logger.info(
   814	                        "Multimodal feature transport auto-resolved to cpu: "
   815	                        "the model has not opted into CUDA VMM transport."
   816	                    )
   817	            else:
   818	                requested_transport = "cpu"
   819	                if is_mnnvl_fabric_device():
   820	                    logger.info(
   821	                        "Multimodal feature transport auto-resolved to cpu: "
   822	                        "GB200/GB300 was detected but no IMEX channel is "
   823	                        "mounted. Configure the MNNVL compute domain or pass "
   824	                        "--mm-feature-transport=cuda_vmm after doing so."
   825	                    )
   826	        else:
   827	            requested_transport = "cpu"
   828	    elif legacy_ipc_is_set and legacy_ipc_enabled != (
   829	        requested_transport == "cuda_ipc"
   830	    ):
   831	        logger.warning(
   832	            "--mm-feature-transport=%s overrides the conflicting legacy "
   833	            "SGLANG_USE_CUDA_IPC_TRANSPORT=%s setting.",
   834	            requested_transport,
   835	            int(legacy_ipc_enabled),
   836	        )
   837
   838	    if cfg.encoder_only and requested_transport in ("cuda_ipc", "cuda_vmm"):
   839	        logger.warning(
   840	            "--mm-feature-transport=%s does not control encoder-only "
   841	            "output transfer; using cpu for this inactive transport. Select "
   842	            "--encoder-transfer-backend for encoder outputs.",
   843	            requested_transport,
   844	        )
   845	        requested_transport = "cpu"
   846
   847	    if requested_transport == "cuda_vmm":
   848	        if not get_platform().is_cuda:
   849	            raise ValueError("--mm-feature-transport=cuda_vmm requires NVIDIA CUDA.")
   850	        if cfg.pp_size != 1:
   851	            raise ValueError(
   852	                "--mm-feature-transport=cuda_vmm does not support pipeline parallelism."
   853	            )
   854	        if envs.SGLANG_RUST_SERVER.get():
   855	            raise ValueError(
   856	                "--mm-feature-transport=cuda_vmm is not supported with "
   857	                "SGLANG_RUST_SERVER."
   858	            )
   859	        pool_budget_mb = envs.SGLANG_MM_FEATURE_CACHE_MB.get()
   860	        handle_kind = "CUDA FABRIC" if cfg.nnodes > 1 else "POSIX FD"
   861	        logger.info(
   862	            "Using CUDA VMM for multimodal features with %s sharing: "
   863	            "reserving up to %d MiB on base GPU %d across %d tokenizer "
   864	            "worker(s). This reduces KV cache headroom; a full pool falls "
   865	            "back to inline CPU transport.",
   866	            handle_kind,
   867	            pool_budget_mb,
   868	            cfg.base_gpu_id,
   869	            cfg.tokenizer_worker_num,
   870	        )
   871
   872	    if requested_transport == "cuda_ipc":
   873	        if not get_platform().is_cuda:
   874	            raise ValueError("--mm-feature-transport=cuda_ipc requires NVIDIA CUDA.")
   875	        if cfg.nnodes != 1:
   876	            raise ValueError(
   877	                "--mm-feature-transport=cuda_ipc only supports a single node."
   878	            )
   879
   880	        pool_budget_mb = envs.SGLANG_MM_FEATURE_CACHE_MB.get()
   881	        logger.info(
   882	            "Using CUDA IPC for multimodal features: reserving up to %d MiB "
   883	            "on base GPU %d across %d tokenizer worker(s). This reduces KV "
   884	            "cache headroom; a full pool falls back to CPU transport.",
   885	            pool_budget_mb,
   886	            cfg.base_gpu_id,
   887	            cfg.tokenizer_worker_num,
   888	        )
   889	        logger.info(
   890	            "CUDA IPC pool-handle caching is %s. It reuses mappings to the "
   891	            "existing bounded pool without reserving another pool; set "
   892	            "SGLANG_USE_IPC_POOL_HANDLE_CACHE=0 to disable it.",
   893	            ("enabled" if envs.SGLANG_USE_IPC_POOL_HANDLE_CACHE.get() else "disabled"),
   894	        )
   895
   896	    declare_resolution(
   897	        server_args,
   898	        "_handle_multimodal_feature_transport",
   899	        mm_feature_transport=requested_transport,
   900	    )
   901	    # The bounded IPC pool owns device residency. Do not retain unpooled
   902	    # tensors after a pool miss, which would make HBM use request-dependent.
   903	    declare_resolution(
   904	        server_args,
   905	        "_handle_multimodal_feature_transport",
   906	        keep_mm_feature_on_device=False,
   907	    )
   908	    envs.SGLANG_USE_CUDA_IPC_TRANSPORT.set(
   909	        "1" if requested_transport == "cuda_ipc" else "0"
   910	    )
